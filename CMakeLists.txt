cmake_minimum_required(VERSION 3.22)
project(Luna VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LUNA_BUILD_QT_SHELL "Build Qt6/QML shell when Qt6 is available" ON)
add_subdirectory(Luna/core)
add_subdirectory(Luna/formats)
add_subdirectory(Luna/services)
add_subdirectory(Luna/runtime)
add_subdirectory(Luna/sdk)
add_subdirectory(Luna/apps)
add_subdirectory(Luna/assets)
add_executable(luna-tests tests/luna_tests.cpp)
target_link_libraries(luna-tests PRIVATE luna_core luna_formats luna_services)
enable_testing()
add_test(NAME luna-core-formats COMMAND luna-tests)
find_package(Qt6 QUIET COMPONENTS Core Gui Qml Quick)
if(LUNA_BUILD_QT_SHELL AND Qt6_FOUND)
  qt_standard_project_setup()
  add_subdirectory(shell)
else()
  message(STATUS "Qt6 Core/Gui/Qml/Quick not found; building native Luna services and tools only")
endif()
